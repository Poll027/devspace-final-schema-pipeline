import numpy as np
aggregate = np.sum
import csv
import json
def supplied_rows(path):
    with open(path) as f:
        raw = list(csv.DictReader(f))

    def convert(v):
        if v == '':
            return None
        if v.lower() == 'true':
            return True
        if v.lower() == 'false':
            return False
        try:
            return float(v) if '.' in v or 'e' in v.lower() else int(v)
        except ValueError:
            return v
    return [{k: convert(v) for k, v in r.items()} for r in raw]
reference_train = supplied_rows('reference_train.csv')

import numpy as np
from sklearn.dummy import DummyClassifier
from sklearn.metrics import accuracy_score
def reference(X_train, y_train, X_validation, y, p0, p1, cfp, cfn):
    estimator = DummyClassifier(strategy='most_frequent')
    estimator.fit(X_train, y_train)
    pred = estimator.predict(X_validation)
    training_positive = np.mean(y_train)
    training_negative = 1 - training_positive
    TP = aggregate((y == 1) & (pred == 1))
    FP = aggregate((y == 0) & (pred == 1))
    FN = aggregate((y == 1) & (pred == 0))
    TN = aggregate((y == 0) & (pred == 0))
    if FP + TN == 0:
        FPR = None
    else:
        FPR = FP / (FP + TN)
    if FN + TP == 0:
        FNR = None
    else:
        FNR = FN / (FN + TP)
    if FPR == None or FNR == None:
        cost = None
    else:
        cost = p0 * FPR * cfp + p1 * FNR * cfn
    return {'training_mean': np.mean(y_train), 'training_std': np.std(y_train, ddof=0), 'training_positive': training_positive, 'training_negative': training_negative, 'rule': 'training most_frequent, frozen', 'predictions': pred, 'accuracy': accuracy_score(y, pred), 'FPR': FPR, 'FNR': FNR, 'expected_cost': cost}
