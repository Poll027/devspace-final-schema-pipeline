import numpy as np
aggregate = np.sum
import csv
import json
def supplied_rows(path):
    with open(path) as f:
        raw = list(csv.DictReader(f))

    def convert(v):
        if v == '':
            return None
        if v.lower() == 'true':
            return True
        if v.lower() == 'false':
            return False
        try:
            return float(v) if '.' in v or 'e' in v.lower() else int(v)
        except ValueError:
            return v
    return [{k: convert(v) for k, v in r.items()} for r in raw]
foundation_outputs = supplied_rows('foundation_outputs.csv')
validation_scores = supplied_rows('validation_scores.csv')
p0 = 0.8
p1 = 0.2
cfp = 1
cfn = 5
bound = 0.3
import numpy as np
def evaluate(ids, output_ids, validation_cohorts, output_cohorts, output_kinds, y, outputs, reference, p0, p1, cfp, cfn, bound):
    ids = np.asarray(ids)
    output_ids = np.asarray(output_ids)
    y = np.asarray(y)
    pred = np.asarray(outputs)
    reference = np.asarray(reference)
    if not np.array_equal(ids, output_ids):
        raise ValueError('observation mismatch')
    if not np.array_equal(np.asarray(validation_cohorts), np.asarray(output_cohorts)):
        raise ValueError('cohort mismatch')
    if aggregate(np.asarray(output_kinds) != 'foundation-model') != 0:
        raise ValueError('output kind mismatch')
    if y.shape != pred.shape or y.shape != reference.shape:
        raise ValueError('shape mismatch')
    TP = aggregate((y == 1) & (pred == 1))
    FP = aggregate((y == 0) & (pred == 1))
    FN = aggregate((y == 1) & (pred == 0))
    TN = aggregate((y == 0) & (pred == 0))
    if FP + TN == 0:
        FPR = None
    else:
        FPR = FP / (FP + TN)
    if FN + TP == 0:
        FNR = None
    else:
        FNR = FN / (FN + TP)
    if FPR == None or FNR == None:
        cost = None
    else:
        cost = p0 * FPR * cfp + p1 * FNR * cfn
    accuracy = aggregate(y == pred) / aggregate((y == 0) | (y == 1))
    baseline_accuracy = aggregate(y == reference) / aggregate((y == 0) | (y == 1))
    baseline_FP = aggregate((y == 0) & (reference == 1))
    baseline_TN = aggregate((y == 0) & (reference == 0))
    baseline_FN = aggregate((y == 1) & (reference == 0))
    baseline_TP = aggregate((y == 1) & (reference == 1))
    if baseline_FP + baseline_TN == 0 or baseline_FN + baseline_TP == 0:
        baseline_cost = None
    else:
        baseline_cost = p0 * baseline_FP / (baseline_FP + baseline_TN) * cfp + p1 * baseline_FN / (baseline_FN + baseline_TP) * cfn
    if cost == None:
        verdict = 'unresolved'
    elif cost <= bound:
        verdict = 'predictive quality accepted'
    else:
        verdict = 'predictive quality rejected'
    return {'aligned': True, 'accuracy': accuracy, 'baseline_accuracy': baseline_accuracy, 'TP': TP, 'FP': FP, 'FN': FN, 'TN': TN, 'FPR': FPR, 'FNR': FNR, 'expected_cost': cost, 'baseline_expected_cost': baseline_cost, 'verdict': verdict, 'limitations': 'Supplied validation cohort only; operating eligibility separate; no universal suitability or protected-test evidence'}

def supplied_mismatch_check():
    try:
        evaluate(ids, output_ids[::-1], validation_cohorts, output_cohorts, output_kinds, y, outputs, reference, p0, p1, cfp, cfn, bound)
    except ValueError:
        return True
    return False
mismatch_rejected = supplied_mismatch_check()
def supplied_rejection_check(cohorts, kinds, message):
    try:
        evaluate(ids, output_ids, validation_cohorts, cohorts, kinds, y, outputs, reference, p0, p1, cfp, cfn, bound)
    except ValueError as e:
        return str(e) == message
    return False
