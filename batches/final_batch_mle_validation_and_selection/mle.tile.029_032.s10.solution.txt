import csv
import json
def supplied_rows(path):
    with open(path) as f:
        raw = list(csv.DictReader(f))
    def convert(v):
        if v == '': return None
        if v.lower() == 'true': return True
        if v.lower() == 'false': return False
        try: return float(v) if '.' in v or 'e' in v.lower() else int(v)
        except ValueError: return v
    return [{k: convert(v) for k,v in r.items()} for r in raw]
paired_trials = supplied_rows('paired_trials.csv')
def supplied_pair(seed, split_random_state, estimator_random_state):
    if split_random_state != seed or estimator_random_state != seed:
        raise ValueError('both random sources must match seed')
    return next(r for r in paired_trials if r['seed'] == seed)
import numpy as np
def paired(declared_seed_list):
    records = []
    selected_results = []
    wins = 0
    for seed in declared_seed_list:
        record = supplied_pair(seed, split_random_state=seed, estimator_random_state=seed)
        records.append(record)
        selected_results.append(record['candidate_result'])
        if record['candidate_result'] < record['baseline_result']:
            wins = wins + 1
    count = len(records)
    return {'records': records, 'completed_pairs': count, 'wins': wins, 'win_probability': wins/count, 'candidate_mean': average(selected_results), 'candidate_std': spread(selected_results, ddof=0), 'limitation': 'Split and estimator randomness controlled; universal deterministic replay not established.'}


import numpy as np
average = np.mean
spread = np.std
declared_seed_list = [r['seed'] for r in paired_trials]
result = paired(declared_seed_list)
