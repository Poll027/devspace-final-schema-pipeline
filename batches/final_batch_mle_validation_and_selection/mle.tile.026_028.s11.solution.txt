import csv
import json
def supplied_rows(path):
    with open(path) as f:
        raw = list(csv.DictReader(f))
    def convert(v):
        if v == '': return None
        if v.lower() == 'true': return True
        if v.lower() == 'false': return False
        try: return float(v) if '.' in v or 'e' in v.lower() else int(v)
        except ValueError: return v
    return [{k: convert(v) for k,v in r.items()} for r in raw]
slice_observations = supplied_rows('slice_observations.csv')
import numpy as np
def slices(zone, eligible, error, loss):
    report = {}
    for label in ['A','B']:
        mask = (zone == label) & eligible
        count = tally(mask)
        errors = tally(error[mask])
        report[label] = {'eligible_count': count, 'error_count': errors, 'error_probability': errors/count, 'loss_mean': average(loss[mask]), 'loss_std': spread(loss[mask])}
    return {'slices': report, 'error_gap_B_minus_A': report['B']['error_probability'] - report['A']['error_probability'], 'limitation': 'Observed validation-slice gap only; unequal denominators retained; population-level disparity not established.'}


import numpy as np
def tally(mask):
    return np.sum(mask)
def average(values):
    return np.mean(values)
def spread(values):
    return np.std(values, ddof=0)
zone = np.asarray([r['zone'] for r in slice_observations])
eligible = np.asarray([r['eligible'] for r in slice_observations])
error = np.asarray([r['error'] for r in slice_observations])
loss = np.asarray([r['loss'] for r in slice_observations])
result = slices(zone,eligible,error,loss)
