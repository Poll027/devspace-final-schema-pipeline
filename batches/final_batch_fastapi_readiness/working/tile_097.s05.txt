# oauthlib implements the authorization-code/PKCE protocol; PyJWT and
# cryptography provide signing. The resource server never owns this key.
import json, secrets, time
from pathlib import Path
from http.server import HTTPServer, BaseHTTPRequestHandler
from urllib.parse import urlsplit, parse_qs
import jwt
from cryptography.hazmat.primitives.asymmetric import rsa
from oauthlib.oauth2 import Server, RequestValidator
cfg=json.loads(Path('issuer_configuration.json').read_text())
key=rsa.generate_private_key(public_exponent=65537,key_size=2048)
kid=secrets.token_hex(16)
jwk=json.loads(jwt.algorithms.RSAAlgorithm.to_jwk(key.public_key()))
jwk.update(kid=kid,use='sig',alg='RS256')
class Validator(RequestValidator):
    def __init__(self): self.codes={}
    def validate_client_id(self,client_id,request,*a,**kw): return client_id==cfg['client_id']
    def get_default_redirect_uri(self,client_id,request,*a,**kw): return cfg['redirect_uri']
    def validate_redirect_uri(self,client_id,redirect_uri,request,*a,**kw): return redirect_uri==cfg['redirect_uri']
    def validate_response_type(self,client_id,response_type,client,request,*a,**kw): return response_type=='code'
    def validate_scopes(self,client_id,scopes,client,request,*a,**kw): return set(scopes)<=set(cfg['scopes'])
    def get_default_scopes(self,client_id,request,*a,**kw): return list(cfg['scopes'])
    def save_authorization_code(self,client_id,code,request,*a,**kw):
        self.codes[code['code']]={'client_id':client_id,'redirect_uri':request.redirect_uri,'scopes':request.scopes,'challenge':request.code_challenge,'method':request.code_challenge_method}
    def is_pkce_required(self,client_id,request): return True
    def client_authentication_required(self,request,*a,**kw): return False
    def authenticate_client_id(self,client_id,request,*a,**kw):
        if client_id!=cfg['client_id']: return False
        request.client=type('RegisteredClient',(),{'client_id':client_id})();return True
    def validate_grant_type(self,client_id,grant_type,client,request,*a,**kw): return grant_type=='authorization_code'

        request.scopes=record['scopes'];request.user='training-principal';return True

class Handler(BaseHTTPRequestHandler):
    def reply(self,status,body,headers=None):
        self.send_response(status)
        for k,v in (headers or {'Content-Type':'application/json'}).items():self.send_header(k,v)
        self.end_headers();self.wfile.write(body.encode())
    def do_GET(self):
        path=urlsplit(self.path).path

import base64, hashlib, secrets
from urllib.parse import urlsplit, parse_qs
import httpx
class AcquisitionHarness:
    def __init__(self,configuration):self.configuration=configuration
    def acquire(self):
        c=self.configuration
        verifier=secrets.token_urlsafe(48)
        challenge=base64.urlsafe_b64encode(hashlib.sha256(verifier.encode()).digest()).rstrip(b'=').decode()
        state=secrets.token_urlsafe(24)
        with httpx.Client(follow_redirects=False) as client:
            redirect=client.get(c['authorization_url'],params={'response_type':'code','client_id':c['client_id'],'redirect_uri':c['redirect_uri'],'scope':' '.join(c['scopes']),'state':state,'code_challenge':challenge,'code_challenge_method':'S256'})
            if redirect.status_code != 302: raise RuntimeError(redirect.text)
            location=redirect.headers['location']
            if location.split('?')[0]!=c['redirect_uri']:raise ValueError('Unregistered callback')
            callback=client.get(location);callback.raise_for_status()
            params=parse_qs(urlsplit(location).query)
            if params.get('state')!=[state]:raise ValueError('State mismatch')
            response=client.post(c['token_url'],data={'grant_type':'authorization_code','client_id':c['client_id'],'code':params['code'][0],'redirect_uri':c['redirect_uri'],'code_verifier':verifier})
            response.raise_for_status()
            return response.json()['access_token']

FILE existing_verifier.py
import jwt
from jwt import PyJWKClient
from fastapi import HTTPException
class ExistingVerifier:
    def __init__(self):self.configuration=None;self.jwks=None
    def configure(self,configuration):
        self.configuration=configuration
        self.jwks=PyJWKClient(configuration['jwks_url'])
    def verify(self,token):
        if self.configuration is None:raise HTTPException(503,'Integration settings incomplete')
        c=self.configuration
        try:
            signing_key=self.jwks.get_signing_key_from_jwt(token)
            return jwt.decode(token,signing_key.key,algorithms=c['algorithms'],issuer=c['issuer'],audience=c['audience'],options={'require':c['required_claims']})
        except jwt.PyJWTError:
            raise HTTPException(401,'Invalid access token')

from fastapi import Request, HTTPException
from fastapi.security import HTTPBearer

configuration=json.loads(Path('issuer_configuration.json').read_text())
existing_verifier=ExistingVerifier();existing_verifier.configure(configuration)
extraction=HTTPBearer(auto_error=False)
async def authenticated(request:Request):
    credential=await extraction(request)
    if credential is None:raise HTTPException(401,'Bearer token required')
    return existing_verifier.verify(credential.credentials)
def enforce_permission(principal,required):
    if required not in principal.get('scope','').split():raise HTTPException(403,'Permission required')
    return principal
from fastapi import Depends
def read_orders(principal=Depends(authenticated)):
    return enforce_permission(principal,'orders:read')
def write_orders(principal=Depends(authenticated)):
    return enforce_permission(principal,'orders:write')
